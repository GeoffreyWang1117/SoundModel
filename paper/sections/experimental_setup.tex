\subsection{Datasets}
\subsubsection{Synthetic Dialogue Data}
\begin{itemize}
    \item 100, 500, 1000 teacher-student dialogues
    \item 6 emotion categories: neutral, happy, sad, angry, fearful, surprised
    \item Generated from templates with emotional context
\end{itemize}

\subsubsection{TTS Synthesis}
\begin{itemize}
    \item Model: XTTS v2 (Coqui)
    \item Speaker: "Claribel Dervla" (consistent voice)
    \item Audio: 16kHz, mono, WAV format
\end{itemize}

\subsection{Model Configuration}
\begin{itemize}
    \item \textbf{Base LLM}: Qwen2.5-1.5B
    \item \textbf{Quantization}: 4-bit (QLoRA)
    \item \textbf{LoRA config}: r=16, alpha=32, dropout=0.05
    \item \textbf{Training}: 5 epochs, batch size 4, lr=5e-5
\end{itemize}

\subsection{Evaluation Metrics}
\begin{itemize}
    \item \textbf{Primary}: Validation perplexity/loss
    \item \textbf{Split}: 80\% train, 20\% validation
    \item \textbf{Reporting}: Best validation loss across epochs
\end{itemize}

\subsection{Implementation Details}
\begin{itemize}
    \item PyTorch 2.5.1, Transformers 4.45.2
    \item NVIDIA GPU (24GB VRAM)
    \item Mixed precision training (fp16)
\end{itemize}
